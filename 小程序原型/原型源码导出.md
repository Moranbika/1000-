```tsx
import { Outlet, useLocation } from 'react-router-dom';
import BottomTabBar from './BottomTabBar';
import { Toaster } from '@/components/ui/sonner';

export function Layout() {
  const location = useLocation();
  // 详情页不显示底部Tab栏
  const isDetailPage = location.pathname.startsWith('/stage/') ||
                       location.pathname.startsWith('/knowledge/');

  return (
    <div className="min-h-screen bg-background">
      <main className={`mx-auto w-full max-w-lg ${!isDetailPage ? 'pb-20' : ''}`}>
        <Outlet />
      </main>
      {!isDetailPage && <BottomTabBar />}
      <Toaster position="top-center" />
    </div>
  );
}
```

        </button>
        <h1 className="flex-1 truncate text-base font-semibold text-foreground">
          {stage.name}营养指导
        </h1>
      </div>

      {/* 阶段头图与简介 */}
      <div className="relative">
        <Image src={stageImage} alt={stage.name} className="h-40 w-full object-cover" />
        <div className="absolute inset-0 bg-gradient-to-t from-background to-transparent" />
        <div className="absolute bottom-0 left-0 right-0 p-4">
          <div className="inline-flex items-center rounded-full px-3 py-1 text-xs font-medium text-white" style={{ backgroundColor: stage.color }}>
            {stage.period}
          </div>
          <h2 className="mt-2 text-xl font-bold text-foreground">{stage.name}</h2>
          <p className="mt-1 text-sm text-muted-foreground">{stage.description}</p>
        </div>
      </div>

      {/* 阶段快速切换（上一阶段/下一阶段） */}
      <div className="mt-4 flex items-center gap-2 px-4">
        {currentIndex > 0 && (
          <button onClick={() => navigate(`/stage/${MOCK_STAGES[currentIndex - 1].id}`)}
            className="flex flex-1 items-center justify-center gap-1 rounded-xl border border-border/60 bg-card px-3 py-2 text-xs text-muted-foreground hover:bg-muted/50">
            <ChevronLeft className="size-4" />
            {MOCK_STAGES[currentIndex - 1].shortName}
          </button>
        )}
        {currentIndex < MOCK_STAGES.length - 1 && (
          <button onClick={() => navigate(`/stage/${MOCK_STAGES[currentIndex + 1].id}`)}
            className="flex flex-1 items-center justify-center gap-1 rounded-xl border border-border/60 bg-card px-3 py-2 text-xs text-muted-foreground hover:bg-muted/50">
            {MOCK_STAGES[currentIndex + 1].shortName}
            <ChevronRight className="size-4" />
          </button>
        )}
      </div>

      {/* 手风琴式6大模块 */}
      <div className="mt-6 space-y-3 px-4">
        {SECTION_CONFIG.map((section, idx) => {
          const Icon = section.icon;
          const isExpanded = expandedSections.has(section.key);
          return (
            <motion.div
              key={section.key}
              initial={{ opacity: 0, y: 15 }}
              animate={{ opacity: 1, y: 0 }}

            >
              <button onClick={() => toggleSection(section.key)}
                className="flex w-full items-center gap-3 px-4 py-3.5 text-left hover:bg-muted/30">
                <div className="flex size-9 items-center justify-center rounded-xl text-white" style={{ backgroundColor: stage.color }}>
                  <Icon className="size-4.5" />
                </div>
                <span className="flex-1 text-sm font-semibold text-foreground">{section.label}</span>
                <ChevronRight className={`size-5 text-muted-foreground transition-transform ${isExpanded ? 'rotate-90' : ''}`} />
              </button>

              {isExpanded && (
                <motion.div
                  initial={{ height: 0, opacity: 0 }}
                  animate={{ height: 'auto', opacity: 1 }}
                  exit={{ height: 0, opacity: 0 }}
                  transition={{ duration: 0.3 }}
                  className="border-t border-border/40 px-4 py-4"
                >
                  {/* 核心营养原则：编号列表 */}
                  {section.key === 'principles' && (
                    <ul className="space-y-3">
                      {stage.corePrinciples.map((p, i) => (
                        <li key={i} className="flex gap-3">
                          <span className="mt-0.5 flex size-5 shrink-0 items-center justify-center rounded-full text-xs font-bold text-white"
                            style={{ backgroundColor: stage.color }}>{i + 1}</span>
                          <p className="text-sm leading-relaxed text-foreground/90">{p}</p>
                        </li>
                      ))}
                    </ul>
                  )}

                  {/* 重点营养素：卡片 + 食物来源标签 */}
                  {section.key === 'nutrients' && (
                    <div className="space-y-4">
                      {stage.keyNutrients.map((n, i) => (
                        <div key={i} className="rounded-xl bg-muted/30 p-3">
                          <div className="flex items-center justify-between">
                            <span className="font-semibold text-foreground">{n.name}</span>
                            <span className="rounded-full px-2 py-0.5 text-xs font-medium"
                              style={{ backgroundColor: `${stage.color}20`, color: stage.color }}>
                              {n.dailyIntake}
                            </span>
                          </div>
                          <div className="mt-2">
                            <p className="text-xs text-muted-foreground">食物来源</p>
                            <div className="mt-1 flex flex-wrap gap-1.5">
                              {n.foodSources.map((src, j) => (
                                <span key={j} className="rounded-md bg-card px-2 py-0.5 text-xs text-foreground/80 border border-border/40">
                                  {src}
                                </span>
                              ))}
                            </div>
                          </div>
                        </div>
                      ))}
                    </div>
                  )}

                  {/* 膳食建议：圆点列表 */}
                  {section.key === 'dietary' && (
                    <ul className="space-y-2.5">
                      {stage.dietaryAdvice.map((d, i) => (
                        <li key={i} className="flex gap-2.5">
                          <span className="mt-2 size-1.5 shrink-0 rounded-full" style={{ backgroundColor: stage.color }} />
                          <p className="text-sm leading-relaxed text-foreground/90">{d}</p>
                        </li>
                      ))}
                    </ul>
                  )}

                  {/* 注意事项：警示卡片 */}
                  {section.key === 'cautions' && (
                    <div className="space-y-2.5">
                      {stage.cautions.map((c, i) => (
                        <div key={i} className="flex gap-2.5 rounded-lg bg-destructive/5 p-2.5">
                          <AlertTriangle className="mt-0.5 size-4 shrink-0 text-destructive" />
                          <p className="text-sm text-destructive/90">{c}</p>
                        </div>
                      ))}
                    </div>
                  )}

                  {/* 一日参考食谱：6餐标签化展示 */}
                  {section.key === 'recipe' && (
                    <div className="space-y-3">
                      {[
                        { key: 'breakfast', label: '早餐' },
                        { key: 'morningSnack', label: '上午加餐' },
                        { key: 'lunch', label: '午餐' },
                        { key: 'afternoonSnack', label: '下午加餐' },
                        { key: 'dinner', label: '晚餐' },
                        { key: 'eveningSnack', label: '夜宵' },
                      ].map((meal) => {
                        const content = stage.sampleRecipe[meal.key as keyof typeof stage.sampleRecipe];
                        if (!content) return null;
                        return (
                          <div key={meal.key} className="flex gap-3 rounded-xl bg-muted/30 p-3">
                            <span className="shrink-0 rounded-md px-2 py-1 text-xs font-medium text-white"
                              style={{ backgroundColor: stage.color }}>{meal.label}</span>
                            <p className="text-sm text-foreground/90">{content}</p>
                          </div>

                      })}

                      {stage.warningSigns.map((w, i) => (
                        <div key={i} className="flex gap-2.5 rounded-lg border border-warning/30 bg-warning/5 p-2.5">

                          <p className="text-sm text-foreground/90">{w}</p>
                        </div>
                      ))}
                      <p className="pt-1 text-xs text-muted-foreground">
                        <BookOpen className="mr-1 inline size-3" />
                        出现以上情况请及时就医，不要自行处理
                      </p>
                    </div>
                  )}
                </motion.div>
              )}
            </motion.div>
          );
        })}
      </div>

      {/* 跳转知识库 */}
      <div className="mt-8 px-4">
        <button onClick={() => navigate('/knowledge')}
          className="flex w-full items-center justify-center gap-2 rounded-2xl bg-muted/50 px-4 py-3 text-sm text-muted-foreground hover:bg-muted">
          <BookOpen className="size-4" />
          查看更多营养知识
          <ChevronRight className="size-4" />
        </button>
      </div>
    </div>
  );
}
